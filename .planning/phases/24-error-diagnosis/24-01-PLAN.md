```csharp
/// <summary>
/// Centralized exit code constants for ChronoView UI Automation CLI.
/// All command handlers should return these values instead of calling Environment.Exit().

public static class ExitCodes

    /// <summary>Command executed successfully</summary>
    public const int SUCCESS = 0;

    /// <summary>General error (exception, automation failure, etc.)</summary>
    public const int ERROR = 1;

    /// <summary>Resource not found (window, file, element, etc.)</summary>
    public const int NOT_FOUND = 2;

    /// <summary>Operation timed out</summary>
    public const int TIMEOUT = 3;

    /// <summary>Invalid command-line arguments</summary>
    public const int INVALID_ARGUMENT = 4;
}
```

Modify Program.cs to add try-catch wrapper around `rootCommand.InvokeAsync(args)` at line 60.

Current code (line 60):
```csharp
return await rootCommand.InvokeAsync(args);
```

Replace with:
```csharp
try
{
    return await rootCommand.InvokeAsync(args);
}
catch (FlaUI.Core.Exceptions.AutomationException ex)
{
    WriteError($"[UI Automation Error] {ex.Message}");
    WriteError($"Command: {string.Join(" ", args)}");
    if (s_isVerbose)
    {
        WriteError($"Stack trace: {ex.StackTrace}");
    }
    return ExitCodes.ERROR;
}
catch (System.CommandLine.Parsing.CommandLineException ex)
{
    // Parse errors are already handled by System.CommandLine, but just in case
    WriteError($"[Command Parse Error] {ex.Message}");
    return ExitCodes.INVALID_ARGUMENT;
}
catch (Exception ex)
{
    WriteError($"[Error] {ex.GetType().Name}: {ex.Message}");
    WriteError($"Command: {string.Join(" ", args)}");
    if (s_isVerbose)
    {
        WriteError($"Stack trace: {ex.StackTrace}");
    }
    return ExitCodes.ERROR;
}
```

Also add at top of file with other usings:
```csharp
using UiAutomation.Commands;
using FlaUI.Core.Exceptions;
```

And add helper method below Main:
```csharp
/// <summary>
/// Writes error message to stderr.
/// </summary>
static void WriteError(string message)
{
    Console.Error.WriteLine(message);
}
```
