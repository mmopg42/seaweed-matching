import pytest
from ChronoViewTestAgent import ChronoViewCLI

class TestConnectivity:
    """Test ChronoView connectivity and basic UI detection."""

    def test_chronoview_is_running(self, cli: ChronoViewCLI):
        """Verify ChronoView main window is accessible."""
        result = cli.run(["test", "connectivity"])
        assert result["success"], f"Connectivity test failed: {result.get('error')}"
        assert result["data"]["windowFound"], "MainWindow not found"

    def test_capabilities_available(self, cli: ChronoViewCLI):
        """Verify all automation capabilities are available."""
        result = cli.run(["test", "capabilities"])
        assert result["success"]
        caps = result["data"]
        assert "windows" in caps
        assert "controllers" in caps
        assert len(caps["windows"]) >= 1  # At least MainWindow

    def test_datagrid_accessible(self, cli: ChronoViewCLI):
        """Verify FileGroupDataGrid is accessible."""
        result = cli.run(["test", "datagrid"])
        assert result["success"]
        assert result["data"]["accessible"]
        assert "rowCount" in result["data"]

import pytest
import time
from ChronoViewTestAgent import ChronoViewCLI

@pytest.mark.order(after="test_connectivity.py")
class TestWorkflow:
    """Test ChronoView workflow automation."""

    def test_start_monitoring_workflow(self, cli: ChronoViewCLI, require_chronoview):
        """Test complete start monitoring workflow."""
        result = cli.run(["scenario", "start-monitoring"])
        assert result["success"], f"Failed to start monitoring: {result.get('error')}"
        assert result["data"]["started"] is True
        assert result["data"]["buttonState"] == "enabled"  # Start button re-enabled

    def test_stop_monitoring_workflow(self, cli: ChronoViewCLI, require_chronoview):
        """Test complete stop monitoring workflow."""
        # First start if not running
        cli.run(["toolbar", "click", "--button", "start"])
        time.sleep(0.5)

        result = cli.run(["toolbar", "click", "--button", "stop"])
        assert result["success"]
        # Verify button states changed

    def test_configure_paths_workflow(self, cli: ChronoViewCLI, require_chronoview):
        """Test complete path configuration workflow."""
        test_paths = {
            "line1-nir": "C:\\\\test\\\\nir",
            "line1-normal": "C:\\\\test\\\\normal",
            "output": "C:\\\\test\\\\output"
        }
        args = []
        for key, value in test_paths.items():
            args.extend([f"--{key}", value])

        result = cli.run(["scenario", "configure-paths"] + args)
        assert result["success"]
        assert result["data"]["configured"] == list(test_paths.keys())
        assert result["data"]["verified"] is True

    def test_read_camera_states(self, cli: ChronoViewCLI, require_chronoview):
        """Test reading camera button states from WorkflowPanel."""
        result = cli.run(["workflow", "camera-states", "--json"])
        assert result["success"]
        states = result["data"]
        assert "cam1" in states or "camera1" in states
```

import pytest
from ChronoViewTestAgent import ChronoViewCLI

@pytest.mark.order(after="test_workflow.py")
class TestFileOperations:
    """Test file operation automation."""

    def test_read_datagrid_rows(self, cli: ChronoViewCLI, require_chronoview):
        """Test reading FileGroup data from DataGrid."""
        result = cli.run(["datagrid", "rows", "--json"])
        assert result["success"]
        assert "rows" in result["data"]
        assert isinstance(result["data"]["rows"], list)

    def test_read_statistics(self, cli: ChronoViewCLI, require_chronoview):
        """Test reading StatisticsPanel data."""
        result = cli.run(["stats", "get", "--json"])
        assert result["success"]
        stats = result["data"]
        assert "totalGroups" in stats or "totalCount" in stats

    def test_scenario_move_groups_by_index(self, cli: ChronoViewCLI, require_chronoview):
        """Test moving file groups scenario by row indices.

        NOTE: This is a destructive test. Ensure test data exists.
        """
        # First check if we have data
        data_result = cli.run(["datagrid", "rows", "--json"])
        if not data_result["success"] or len(data_result["data"]["rows"]) == 0:
            pytest.skip("No data in DataGrid to test move operation")

        # Move first row (destructive - may need test data setup)
        result = cli.run(["scenario", "move-groups", "--row-indices", "0"])
        assert result["success"]
        assert result["data"]["moved"] >= 0

    def test_batch_export_all_data(self, cli: ChronoViewCLI, require_chronoview):
        """Test batch export all ChronoView data."""
        result = cli.run(["batch", "export-all", "--json"])
        assert result["success"]
        data = result["data"]
        assert "timestamp" in data
        assert "statistics" in data
        assert "dataGrid" in data
        assert "cameraStates" in data
```

These tests verify:
- Data reading operations (non-destructive)
- Move operation workflow (needs test data)
- Batch export functionality
