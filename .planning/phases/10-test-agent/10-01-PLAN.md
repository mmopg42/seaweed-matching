```python
import subprocess
import json
from pathlib import Path
from typing import Optional, Dict, Any

class ChronoViewCLI:
    """Wrapper for ui_automation.exe CLI commands."""

    def __init__(self, exe_path: Optional[str] = None):
        self.exe_path = exe_path or self._find_exe()

    def _find_exe(self) -> str:
        """Find ui_automation.exe in default build location."""
        # Check: ../../skills_scripts/ui_automation/bin/Debug/net10.0/ui_automation.exe
        default = Path(__file__).parent.parent.parent / "skills_scripts" / "ui_automation" / "bin" / "Debug" / "net10.0" / "ui_automation.exe"
        if default.exists():
            return str(default)
        raise FileNotFoundError(f"ui_automation.exe not found at {default}")

    def run(self, args: list, timeout: int = 30) -> Dict[str, Any]:
        """Run CLI command and return parsed JSON result.

        Returns dict with: success (bool), data (dict), error (str), exitCode (int)
        """
        cmd = [self.exe_path] + args + ["--json"]
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            timeout=timeout
        )

        output = result.stdout.strip()
        try:
            response = json.loads(output) if output else {}
        except json.JSONDecodeError:
            response = {"success": False, "error": output or "No output"}

        response["exitCode"] = result.returncode
        return response
```

```python
import pytest

# Pydantic models for output validation
from pydantic import BaseModel

class SuccessResponse(BaseModel):
    success: bool
    data: Dict[str, Any]

class ErrorResponse(BaseModel):
    success: bool
    error: str
    errorCode: int

@pytest.fixture(scope="session")
def cli():
    """Shared CLI instance for all tests."""
    return ChronoViewCLI()

@pytest.fixture(scope="function")
def require_chronoview(cli):
    """Skip test if ChronoView is not running."""
    result = cli.run(["test", "connectivity"])
    if not result.get("success") or not result.get("data", {}).get("connected"):
        pytest.skip("ChronoView not running")
    yield cli
```
