# 탭별 샘플 이동 설정 동적 변경 기능 조사 리포트

## 1. 개요

크로노 뷰의 좌측 패널에 있는 "샘플 이동 설정"을 현재 활성화된 탭(Line1/Line2/Combined)에 따라 동적으로 변경하는 기능 구현을 위한 조사 리포트입니다.

### 요구사항 요약
- **Line1 탭 활성화**: "Line1 이동 설정"으로 변경, Line1 전용 시료명/NIR 이동/전체 이동 값 표시
- **Line2 탭 활성화**: "Line2 이동 설정"으로 변경, Line2 전용 시료명/NIR 이동/전체 이동 값 표시
- **Combined 탭 활성화**: "통합 이동 설정"으로 변경, 2컬럼 레이아웃(좌: Line1, 우: Line2)으로 양쪽 라인 설정 동시 표시

    private async Task SaveSampleMoveSettingsAsync()
    {
        try
        {
            lock (_saveLock)
            {
                _saveDebounceTimer?.Stop();
                _saveDebounceTimer = new System.Windows.Threading.DispatcherTimer
                {
                    Interval = TimeSpan.FromMilliseconds(500)
                };
                _saveDebounceTimer.Tick += async (s, e) =>
                {
                    _saveDebounceTimer.Stop();
                    try
                    {
                        var config = await _configManager.LoadConfigurationAsync<ApplicationConfiguration>();
                        
                        // SampleName 저장
                        config.MatchingSettings.SampleName = string.IsNullOrWhiteSpace(SampleName) ? null : SampleName;
                        
                        // MoveNir 저장 (빈 값 또는 파싱 실패 시 null)
                        config.MatchingSettings.MoveNir = string.IsNullOrWhiteSpace(MoveNir) 
                            ? null 
                            : (int.TryParse(MoveNir, out int nirValue) ? nirValue : null);
                        
                        // MoveAllData 저장 (빈 값 또는 파싱 실패 시 null)
                        config.MatchingSettings.MoveAllData = string.IsNullOrWhiteSpace(MoveAllData) 
                            ? null 
                            : (int.TryParse(MoveAllData, out int allValue) ? allValue : null);
                        
                        await _configManager.SaveConfigurationAsync(config);
                        _logger.LogDebug("Sample move settings saved: SampleName={SampleName}, MoveNir={MoveNir}, MoveAllData={MoveAllData}", 
                            SampleName, MoveNir, MoveAllData);
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "Failed to save sample move settings");
                    }
                };
                _saveDebounceTimer.Start();
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error setting up save debounce timer");
        }
    }

    public async Task ExecuteMoveAsync(IEnumerable<FileGroupViewModel> selectedGroups, string moveNirCount, string moveAllDataCount, string? subject = null)
    {
        if (IsOperationInProgress) return;
        try {
            IsOperationInProgress = true;

            await SaveLimitsAsync(moveNirCount, moveAllDataCount);

            var config = await _configManager.LoadConfigurationAsync<ApplicationConfiguration>();
            
            // Validate output path before proceeding
            var outputPath = config.MatchingSettings.OutputPath;
            
            // If output path is empty, try to use default path based on BasePath
            if (string.IsNullOrWhiteSpace(outputPath))
            {
                var basePath = !string.IsNullOrWhiteSpace(config.BasePath) ? config.BasePath : "D:/Data";
                outputPath = System.IO.Path.Combine(basePath, "Output");
                _logger.LogInformation("OutputPath is empty, using default: {Path}", outputPath);
                
                // Ask user if they want to use default path or cancel
                var message = $"출력 경로가 설정되지 않았습니다.\n\n기본 경로를 사용하시겠습니까?\n{outputPath}\n\n(아니오를 선택하면 이동 작업이 취소됩니다.)";
                var dialogResult = System.Windows.MessageBox.Show(message, "출력 경로 미지정", System.Windows.MessageBoxButton.YesNo, System.Windows.MessageBoxImage.Question);
                
                if (dialogResult == System.Windows.MessageBoxResult.No)
                {
                    var errorMessage = "출력 경로가 설정되지 않아 이동 작업을 취소했습니다.\n설정 다이얼로그에서 출력 경로를 지정해주세요.";
                    _logger.LogWarning("Move operation cancelled by user: OutputPath is empty");
                    LogRequested?.Invoke(LogSeverity.Warning, "FileOperation", errorMessage);
                    StatusChanged?.Invoke("이동 취소: 출력 경로 미지정");
                    return;
                }
                
                // Save the default path to configuration for future use
                config.MatchingSettings.OutputPath = outputPath;
                await _configManager.SaveConfigurationAsync(config);
                _logger.LogInformation("Saved default OutputPath to configuration: {Path}", outputPath);
            }

            // Check if output directory exists, if not, try to create it
            if (!System.IO.Directory.Exists(outputPath))
            {
                try
                {
                    System.IO.Directory.CreateDirectory(outputPath);
                    _logger.LogInformation("Created output directory: {Path}", outputPath);
                }
                catch (Exception ex)
                {
                    var errorMessage = $"출력 경로를 생성할 수 없습니다.\n경로: {outputPath}\n오류: {ex.Message}";
                    _logger.LogError(ex, "Failed to create output directory: {Path}", outputPath);
                    LogRequested?.Invoke(LogSeverity.Error, "FileOperation", errorMessage);
                    StatusChanged?.Invoke("이동 실패: 출력 경로 생성 불가");
                    System.Windows.MessageBox.Show(errorMessage, "이동 실패", System.Windows.MessageBoxButton.OK, System.Windows.MessageBoxImage.Error);
                    return;
                }
            }

            int.TryParse(moveNirCount, out int nirLimit);
            int.TryParse(moveAllDataCount, out int totalLimit);

            var result = await _moveService.BatchMoveAsync(
                selectedGroups.Select(g => g.Model),
                outputPath,
                totalLimit, nirLimit, subject,
                new Progress<OperationProgress>(p => { ProgressValue = (int)p.PercentComplete; StatusChanged?.Invoke(p.Status); }));

            LogRequested?.Invoke(LogSeverity.Info, "FileOperation", result.Success ? "Move completed successfully." : "Move partial/failed.");
        } finally { IsOperationInProgress = false; }
    }

#### 디바운스 타이머 세부사항 (명시)
- `SaveLineSettingsAsync(int lineNumber)`는 **라인별 타이머(2개)**가 아니라 **단일 디바운스 타이머(1개)를 공유**하는 방식으로 구현을 권장합니다.
  - **이유**: 구현 단순화(타이머/이벤트/락 관리 1세트), 저장 경쟁 감소, 디스크 쓰기 횟수 감소
- 동작 방식:
  - Line1/Line2 어느 쪽 입력이 바뀌어도 `SaveLineSettingsAsync(...)`가 호출되며, 내부에서 **같은 타이머를 Stop→Start로 리셋**
  - 마지막 입력 이후 \(예: 500ms\) 추가 변경이 없으면 Tick에서 **Line1/Line2 설정을 모두 config에 반영하고 한 번에 저장**

예시(개념 코드):

```csharp
private readonly object _saveLock = new();
private System.Windows.Threading.DispatcherTimer? _lineSettingsSaveDebounceTimer;

private Task SaveLineSettingsAsync(int lineNumber)
{
    // lineNumber는 호출 원인을 나타낼 뿐, 단일 저장(양 라인 동시 반영)을 사용하면 "저장 범위"를 라인으로 제한할 필요가 없습니다.
    lock (_saveLock)
    {
        _lineSettingsSaveDebounceTimer?.Stop();
        _lineSettingsSaveDebounceTimer = new System.Windows.Threading.DispatcherTimer
        {
            Interval = TimeSpan.FromMilliseconds(500)
        };
        _lineSettingsSaveDebounceTimer.Tick += async (s, e) =>
        {
            _lineSettingsSaveDebounceTimer.Stop();
            var config = await _configManager.LoadConfigurationAsync<ApplicationConfiguration>();

            config.MatchingSettings.Line1Settings.SampleName = string.IsNullOrWhiteSpace(Line1SampleName) ? null : Line1SampleName;
            config.MatchingSettings.Line1Settings.MoveNir = string.IsNullOrWhiteSpace(Line1MoveNir) ? null : (int.TryParse(Line1MoveNir, out var n1) ? n1 : null);
            config.MatchingSettings.Line1Settings.MoveAllData = string.IsNullOrWhiteSpace(Line1MoveAllData) ? null : (int.TryParse(Line1MoveAllData, out var a1) ? a1 : null);

            config.MatchingSettings.Line2Settings.SampleName = string.IsNullOrWhiteSpace(Line2SampleName) ? null : Line2SampleName;
            config.MatchingSettings.Line2Settings.MoveNir = string.IsNullOrWhiteSpace(Line2MoveNir) ? null : (int.TryParse(Line2MoveNir, out var n2) ? n2 : null);
            config.MatchingSettings.Line2Settings.MoveAllData = string.IsNullOrWhiteSpace(Line2MoveAllData) ? null : (int.TryParse(Line2MoveAllData, out var a2) ? a2 : null);

            await _configManager.SaveConfigurationAsync(config);
        };
        _lineSettingsSaveDebounceTimer.Start();
    }

    return Task.CompletedTask;
}
```

### 3.5 삭제 실행 로직 변경 (탭 기반)

레거시 `SampleName`은 삭제(Quarantine) 경로에도 반영되어 왔기 때문에, 레거시 제거 후에도 **활성 탭 기준으로 라인별 시료명(subject)을 사용**하도록 삭제 플로우를 분기해야 합니다.

#### ExecuteDeleteWithConfirmation 수정(권장)
- Line1 탭: `Dashboard.Line1Groups` 대상 + `Line1SampleName`을 subject로 전달
- Line2 탭: `Dashboard.Line2Groups` 대상 + `Line2SampleName`을 subject로 전달
- Combined 탭: Line1 → Line2 순차 실행(이동과 동일한 원칙) + 각 라인의 시료명을 subject로 전달

namespace ChronoView.Core.LineMove;

/// <summary>
/// 라인별 이동 작업 실행기
/// </summary>
public interface ILineMoveExecutor
{
    Task<OperationResult> ExecuteAsync(LineMoveContext context, IProgress<OperationProgress>? progress = null, CancellationToken ct = default);
}

public class LineMoveExecutor : ILineMoveExecutor
{
    private readonly IMoveService _moveService;
    private readonly ILogger<LineMoveExecutor> _logger;

    public LineMoveExecutor(IMoveService moveService, ILogger<LineMoveExecutor> logger)
    {
        _moveService = moveService;
        _logger = logger;
    }

    public async Task<OperationResult> ExecuteAsync(LineMoveContext context, IProgress<OperationProgress>? progress = null, CancellationToken ct = default)
    {
        _logger.LogInformation("Executing move for Line {LineNumber} with {GroupCount} groups", 
            context.LineNumber, context.Groups.Count());

        int.TryParse(context.MoveNir?.ToString() ?? "", out int nirLimit);
        int.TryParse(context.MoveAllData?.ToString() ?? "", out int totalLimit);

        return await _moveService.BatchMoveAsync(
            context.Groups,
            context.OutputPath,
            totalLimit,
            nirLimit,
            context.SampleName,
            progress,
            ct);
    }
}
